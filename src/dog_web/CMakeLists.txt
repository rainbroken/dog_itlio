cmake_minimum_required(VERSION 3.16)
project(dog_web LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(dog_terrain REQUIRED)
find_package(dog_navigation REQUIRED)
find_package(Boost REQUIRED COMPONENTS system)
find_package(Threads REQUIRED)
find_package(nlohmann_json REQUIRED)
foreach(dependency rclcpp rclcpp_action dog_interfaces sensor_msgs nav_msgs geometry_msgs diagnostic_msgs std_msgs std_srvs tf2_ros)
  find_package(${dependency} REQUIRED)
endforeach()
add_executable(planning_preview src/planning_preview.cpp src/localization_bridge.cpp src/navigation_bridge.cpp)
ament_target_dependencies(planning_preview dog_navigation)
add_library(control_socket STATIC src/control_socket.cpp)
target_include_directories(control_socket PUBLIC src)
target_link_libraries(control_socket PUBLIC Boost::system Threads::Threads nlohmann_json::nlohmann_json)
add_library(motion_settings STATIC src/motion_settings.cpp)
add_library(resource_monitor STATIC src/resource_monitor.cpp)
target_include_directories(resource_monitor PUBLIC src)
target_link_libraries(resource_monitor PUBLIC nlohmann_json::nlohmann_json)
add_library(cloud_preview STATIC src/cloud_preview.cpp src/cloud_tiles.cpp src/pcd_catalog.cpp)
add_library(map_library STATIC src/map_library.cpp)
target_include_directories(map_library PUBLIC src)
target_link_libraries(map_library PUBLIC cloud_preview dog_terrain::dog_terrain)
target_include_directories(cloud_preview PUBLIC src)
ament_target_dependencies(cloud_preview rclcpp sensor_msgs tf2_ros)
target_link_libraries(cloud_preview nlohmann_json::nlohmann_json)
target_include_directories(motion_settings PUBLIC src)
target_link_libraries(motion_settings PUBLIC nlohmann_json::nlohmann_json)
ament_target_dependencies(planning_preview ament_index_cpp rclcpp rclcpp_action dog_interfaces sensor_msgs
  nav_msgs geometry_msgs diagnostic_msgs std_msgs std_srvs tf2_ros)
target_link_libraries(planning_preview dog_terrain::dog_terrain Boost::system Threads::Threads motion_settings cloud_preview control_socket map_library resource_monitor)
target_compile_options(planning_preview PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS planning_preview DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch web DESTINATION share/${PROJECT_NAME}
  PATTERN node_modules EXCLUDE PATTERN tools EXCLUDE PATTERN package.json EXCLUDE
  PATTERN package-lock.json EXCLUDE)
