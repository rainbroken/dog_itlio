cmake_minimum_required(VERSION 3.8)
project(dog_bringup)
find_package(ament_cmake REQUIRED)
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME}
  PATTERN __pycache__ EXCLUDE PATTERN "*.pyc" EXCLUDE)
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(test_bringup test/test_bringup.py TIMEOUT 60)
  ament_add_pytest_test(test_configuration test/test_configuration.py TIMEOUT 60)
  ament_add_pytest_test(test_navigation_bringup test/test_navigation_bringup.py TIMEOUT 100)
endif()
ament_package()
