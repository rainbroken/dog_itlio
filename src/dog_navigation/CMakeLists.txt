cmake_minimum_required(VERSION 3.16)
project(dog_navigation LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(dog_terrain REQUIRED)
find_package(PCL REQUIRED COMPONENTS common kdtree)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(dog_interfaces REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(pcl_conversions REQUIRED)
add_library(local_planner SHARED
  src/local_planner/obstacle_memory.cpp src/local_planner/collision_overlay.cpp
  src/local_planner/local_path.cpp src/global_planner/terrain_planner.cpp
  src/decision/localization_gate.cpp src/global_planner/terrain_route_validation.cpp)
target_include_directories(local_planner PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include> ${PCL_INCLUDE_DIRS})
target_link_libraries(local_planner PUBLIC dog_terrain::dog_terrain ${PCL_LIBRARIES})
target_compile_options(local_planner PRIVATE -Wall -Wextra -Wpedantic)
add_executable(local_navigation src/ros/local_navigation_node.cpp src/ros/navigation_parameters.cpp
  src/ros/navigation_input.cpp src/ros/navigation_processing.cpp src/ros/navigation_output.cpp)
target_link_libraries(local_navigation local_planner)
ament_target_dependencies(local_navigation rclcpp dog_interfaces nav_msgs sensor_msgs geometry_msgs std_msgs std_srvs tf2_ros pcl_conversions)
target_compile_options(local_navigation PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS local_planner LIBRARY DESTINATION lib)
install(TARGETS local_navigation DESTINATION lib/${PROJECT_NAME})
add_library(navigation_execution SHARED src/local_planner/timed_trajectory.cpp
  src/local_planner/path_tracker.cpp src/decision/task_supervisor.cpp src/decision/safety_arbiter.cpp)
target_link_libraries(navigation_execution PUBLIC local_planner)
add_executable(navigation_executor src/execution/executor_node.cpp src/execution/executor_inputs.cpp
  src/execution/executor_tick.cpp src/execution/executor_output.cpp)
target_link_libraries(navigation_executor navigation_execution)
ament_target_dependencies(navigation_executor rclcpp rclcpp_action dog_interfaces nav_msgs geometry_msgs std_msgs std_srvs tf2_ros)
add_executable(velocity_guard src/execution/velocity_guard_node.cpp src/execution/guard_checks.cpp)
target_link_libraries(velocity_guard navigation_execution)
ament_target_dependencies(velocity_guard rclcpp dog_interfaces nav_msgs geometry_msgs std_msgs std_srvs tf2_ros)
install(TARGETS navigation_execution LIBRARY DESTINATION lib)
install(TARGETS navigation_executor velocity_guard DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
ament_export_include_directories(include)
ament_export_dependencies(dog_terrain)
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME}
  PATTERN "__pycache__" EXCLUDE PATTERN "*.pyc" EXCLUDE)
install(FILES README.md EXECUTION.md PERFORMANCE.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  add_executable(benchmark_navigation test/benchmark_navigation.cpp test/benchmark_control.cpp test/benchmark_perception.cpp test/benchmark_planner.cpp)
  target_link_libraries(benchmark_navigation navigation_execution)
  find_package(ament_cmake_gtest REQUIRED)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_gtest(test_local_planner test/test_local_planner.cpp TIMEOUT 60)
  ament_add_gtest(test_execution test/test_execution.cpp TIMEOUT 60)
  ament_add_gtest(test_navigation_motion_settings test/test_navigation_motion_settings.cpp TIMEOUT 15)
  target_link_libraries(test_navigation_motion_settings navigation_execution)
  target_link_libraries(test_execution navigation_execution)
  target_link_libraries(test_local_planner local_planner)
  ament_add_pytest_test(test_local_navigation test/test_local_navigation.py TIMEOUT 90)
  ament_add_pytest_test(test_execution_runtime test/test_execution_runtime.py TIMEOUT 90)
endif()
ament_package()
